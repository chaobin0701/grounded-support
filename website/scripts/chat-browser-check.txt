async (page) => {
  const context = await page.context().browser().newContext();
  const p = await context.newPage();
  const base = 'http://localhost:8849';
  const checks = [];
  const check = (ok, name) => { if (!ok) throw new Error(name); checks.push(name); };
  const id = '11111111-1111-4111-8111-111111111111';
  const now = new Date().toISOString();
  let created = 0, removed = 0, recovery = false, polls = 0;
  let rows = Array.from({length: 35}, (_, n) => ({ id: `22222222-2222-4222-8222-${String(n).padStart(12,'0')}`, title: `历史会话 ${n}`, summary: '', createdAt: now, updatedAt: now }));
  let messages = [];
  const citation = {chunkId: 1, documentId: 1, documentTitle: '产品资料', originalFilename: '产品资料.md'};
  await context.addInitScript(() => {
    const originalFetch = window.fetch.bind(window);
    window.chatTest = { mode: 'stream', streams: 0, requestId: '', release: null };
    window.fetch = async (input, options) => {
      if (String(input).endsWith('/messages/stream')) {
        window.chatTest.streams++;
        window.chatTest.requestId = options.headers['Idempotency-Key'];
        if (window.chatTest.mode === 'network') throw new TypeError('Simulated connection loss');
        if (window.chatTest.mode === '429') return new Response(JSON.stringify({message:'限流'}), {status:429});
        if (window.chatTest.mode === '409') return new Response('event: error\ndata: {"status":409,"message":"上一轮处理中"}\n\n', {headers:{'Content-Type':'text/event-stream'}});
        const encoder = new TextEncoder();
        return new Response(new ReadableStream({start(controller) {
          const emit = (event, value) => controller.enqueue(encoder.encode(`event: ${event}\ndata: ${JSON.stringify(value)}\n\n`));
          emit('answer_started', {citations:[]});
          emit('progress', {type:'progress', id:'tool-start', stage:'tool', status:'running', text:'正在查询产品资料…'});
          emit('answer_delta', {text:'前半段回答'});
          window.chatTest.release = () => {
            emit('progress', {type:'progress', id:'tool-completed', stage:'tool', status:'completed', text:'资料查询完成', resultCount:1, durationMs:200});
            emit('answer_delta', {text:'，后半段回答。'});
            emit('done', {answer:'前半段回答，后半段回答。', refused:false, citations:[]});
            controller.close();
          };
        }}), {headers:{'Content-Type':'text/event-stream'}});
      }
      return originalFetch(input, options);
    };
  });
  await context.route('**/api/chat/**', async route => {
    const req = route.request(), path = new URL(req.url()).pathname;
    let data;
    if (path.endsWith('/presentation')) data = {title:'', description:'', suggestedQuestions:[]};
    else if (path.endsWith('/conversations')) {
      if (req.method() === 'POST') { created++; const item = {id,title:'测试问题',summary:'',createdAt:now,updatedAt:now}; rows.unshift(item); data=item; }
      else data=rows;
    } else if (path.endsWith('/messages')) data=messages;
    else if (path.endsWith('/runs')) {
      if (!recovery) data=[];
      else {
        polls++;
        const requestId=await p.evaluate(()=>window.chatTest.requestId);
        data=[{id:'33333333-3333-4333-8333-333333333333',clientRequestId:requestId,question:'恢复测试',status:polls===1?'running':'completed',expiresAt:new Date(Date.now()+60000).toISOString()}];
      }
    } else if (req.method()==='DELETE') { removed++; rows=rows.filter(row=>!path.endsWith(row.id)); await route.fulfill({status:204}); return; }
    else { await route.fulfill({status:404, contentType:'application/json', body:'{}'}); return; }
    await route.fulfill({contentType:'application/json',body:JSON.stringify(data)});
  });
  try {
    await p.goto(base);
    await p.getByRole('button',{name:'在线咨询',exact:true}).click();
    await p.getByRole('textbox',{name:'输入问题'}).waitFor();
    await p.getByRole('button',{name:'新建会话',exact:true}).click();
    check(created===0,'打开与新建不创建服务端空会话');
    check(await p.getByRole('complementary',{name:'历史会话'}).count()===0,'Widget 无历史切换栏');
    await p.getByRole('textbox',{name:'输入问题'}).fill('测试问题');
    await p.getByRole('button',{name:'发送问题'}).click();
    await p.getByText('前半段回答',{exact:true}).waitFor();
    check(await p.getByRole('button',{name:'新建会话',exact:true}).isDisabled(),'生成中禁用新建');
    await p.getByRole('textbox',{name:'输入问题'}).fill('下一轮草稿');
    await p.getByRole('button',{name:'关闭聊天窗口'}).click();
    await p.getByRole('link',{name:'产品中心',exact:true}).first().click();
    await p.getByRole('button',{name:'打开禾序 AI 助手'}).click();
    await p.getByText('前半段回答',{exact:true}).waitFor();
    await p.getByRole('link',{name:'完整助手',exact:true}).click();
    await p.waitForURL(`${base}/assistant`);
    await p.getByRole('dialog').waitFor({state:'hidden'});
    await p.getByText('前半段回答',{exact:true}).waitFor();
    check(p.url()===`${base}/assistant` && await p.getByRole('dialog').count()===0,'进入同站完整页自动收起窗口');
    check(await p.getByRole('button',{name:'打开禾序 AI 助手'}).count()===0,'完整页不显示机器人');
    check(await p.evaluate(()=>window.chatTest.streams===1),'关闭/换页/进入完整助手不重复请求');
    await p.evaluate(()=>window.chatTest.release());
    await p.getByRole('button',{name:'复制回答'}).waitFor();
    check((await p.locator('.chat-messages').innerText()).includes('前半段回答，后半段回答。'),'同一流继续更新后半段');
    check(await p.getByRole('textbox',{name:'输入问题'}).inputValue()==='下一轮草稿','生成中输入下一轮草稿不丢失');
    check(await p.locator('details.chat-process').evaluateAll(nodes=>nodes.every(n=>!n.open)),'公开处理过程默认折叠');
    messages=[{id:1,role:'user',content:'测试问题',refused:null,citations:[],createdAt:now},{id:2,role:'assistant',content:'## 产品规格\n\n- **额定功率**：1500 W\n- **档位**：三档\n\n**价格和适用面积**：尚未提供。\n\n| 参数 | 值 |\n| --- | --- |\n| 功率 | 1500 W |\n\n```text\n示例代码\n```\n\n'+ '长回答。\n'.repeat(200)+'\n\n<img src=x onerror="window.chatXss=1">\n\n[危险](javascript:alert%281%29)',refused:false,citations:[citation],createdAt:now}];
    await p.reload();
    await p.getByRole('button',{name:'复制回答'}).waitFor();
    check(await p.getByRole('textbox',{name:'输入问题'}).inputValue()==='下一轮草稿','刷新恢复当前会话与草稿');
    check(await p.evaluate(()=>window.chatTest.streams===0),'刷新只读取记录不重新提问');
    check(await p.locator('.chat-sources').count()>0,'引用以来源行展示');
    check(await p.locator('.chat-markdown strong').count()===3,'中文加粗正确渲染，不展示双星号');
    check(await p.locator('.chat-markdown :is(ul,table,pre)').count()===3,'Markdown 列表、表格与代码块正确呈现');
    check(await p.locator('.chat-markdown img,.chat-markdown script,.chat-markdown [href^="javascript:"]').count()===0 && !await p.evaluate(()=>window.chatXss),'Markdown 危险内容不可执行');
    check(await p.locator('.chat-message--user .chat-avatar svg').count()===1 && await p.locator('.chat-message--assistant .chat-avatar svg').count()===1,'用户与 AI 分别显示人物/机器人头像');
    const before = await p.locator('.chat-composer').boundingBox();
    const messagesBefore = await p.locator('.chat-messages').boundingBox();
    await p.getByRole('button',{name:'最近会话',exact:true}).click();
    check(await p.locator('.chat-recent-row').count()===10,'悬浮列表最多显示最近 10 条');
    check(JSON.stringify(before)===JSON.stringify(await p.locator('.chat-composer').boundingBox()) && JSON.stringify(messagesBefore)===JSON.stringify(await p.locator('.chat-messages').boundingBox()),'展开列表不移动聊天正文或输入框');
    await p.screenshot({path:'output/playwright/hexu-assistant-history-open.png'});
    await p.getByRole('button',{name:'查看更多会话',exact:true}).click();
    await p.getByRole('dialog',{name:'全部会话'}).waitFor();
    check(await p.locator('.chat-all-row').count()===36,'查看更多可浏览全部当前历史');
    await p.getByRole('searchbox',{name:'搜索会话'}).fill('测试问题');
    check(await p.locator('.chat-all-row').count()===1,'全部会话支持搜索');
    await p.keyboard.press('Escape');
    await p.getByRole('dialog',{name:'全部会话'}).waitFor({state:'hidden'});
    for (const width of [375,768,1440]) {
      await p.setViewportSize({width,height:900});
      const layout=await p.evaluate(()=>({
        noHorizontal:document.documentElement.scrollWidth<=innerWidth,
        noPageScroll:document.documentElement.scrollHeight<=innerHeight+1,
        input:document.querySelector('.chat-composer').getBoundingClientRect().bottom<=innerHeight,
        footer:document.querySelector('footer').getBoundingClientRect().bottom<=innerHeight+1,
        messages:document.querySelector('.chat-messages').scrollHeight>document.querySelector('.chat-messages').clientHeight,
      }));
      check(Object.values(layout).every(Boolean),`${width}px 完整页无外滚动/横溢出，消息独立滚动，输入及Footer可见`);
      await p.getByRole('button',{name:'最近会话',exact:true}).click();
      check(await p.locator('.chat-history-popover').evaluate(n=>{ const r=n.getBoundingClientRect();return r.left>=0&&r.right<=innerWidth&&r.bottom<=innerHeight; }),`${width}px 悬浮会话列表在视口内`);
      await p.keyboard.press('Escape');
      await p.screenshot({path:`output/playwright/hexu-assistant-${width}.png`});
    }
    await p.setViewportSize({width:375,height:812});
    await p.getByRole('link',{name:'禾序智能首页'}).click();
    await p.getByRole('button',{name:'打开禾序 AI 助手'}).click();
    await p.getByRole('dialog').waitFor();
    check(await p.evaluate(()=>document.body.style.overflow==='hidden'),'手机弹窗锁定背景滚动');
    check(await p.getByRole('dialog').evaluate(n=>{const r=n.getBoundingClientRect();return r.left>=0&&r.right<=innerWidth&&r.top>=0&&r.bottom<=innerHeight;}),'手机窗口适配视口');
    await p.getByRole('button',{name:'发送问题'}).focus();
    await p.keyboard.press('Tab');
    check(await p.getByRole('button',{name:'新建会话',exact:true}).evaluate(n=>n===document.activeElement),'手机 Tab 焦点限制在窗口');
    await p.getByRole('dialog').evaluate(async n=>{ await Promise.all(n.getAnimations({subtree:true}).map(a=>a.finished.catch(()=>{}))); });
    await p.screenshot({path:'output/playwright/hexu-widget-mobile.png'});
    await p.keyboard.press('Escape');
    check(await p.getByRole('button',{name:'打开禾序 AI 助手'}).evaluate(n=>n===document.activeElement),'Escape 收起并恢复入口焦点');
    check(await p.evaluate(()=>document.body.style.overflow!=='hidden'),'收起后恢复背景滚动');
    await p.getByRole('button',{name:'打开禾序 AI 助手'}).click();
    await p.evaluate(()=>window.chatTest.mode='429');
    await p.getByRole('textbox',{name:'输入问题'}).fill('限流测试');
    await p.getByRole('button',{name:'发送问题'}).click();
    await p.getByText('提问有些频繁，请稍后再试。',{exact:true}).waitFor();
    check(await p.getByRole('textbox',{name:'输入问题'}).inputValue()==='限流测试','限流保留问题且解除生成状态');
    await p.evaluate(()=>window.chatTest.mode='network');
    await p.getByRole('textbox',{name:'输入问题'}).fill('断网测试');
    await p.getByRole('button',{name:'发送问题'}).click();
    await p.getByText('连接中断或请求超时，请检查上一轮状态。',{exact:true}).waitFor();
    await p.getByRole('button',{name:'新建会话',exact:true}).click({trial:true});
    check(await p.getByRole('button',{name:'新建会话',exact:true}).isEnabled(),'连接中断停止加载且可新建会话');
    check(await p.getByRole('textbox',{name:'输入问题'}).inputValue()==='断网测试','连接中断保留问题');
    recovery=true;
    messages.push({id:3,role:'user',content:'恢复测试',refused:null,citations:[],createdAt:now},{id:4,role:'assistant',content:'恢复后的回答',refused:false,citations:[],createdAt:now});
    await p.evaluate(()=>window.chatTest.mode='409');
    await p.getByRole('textbox',{name:'输入问题'}).fill('恢复测试');
    await p.getByRole('button',{name:'发送问题'}).click();
    await p.getByText('恢复后的回答',{exact:true}).waitFor({timeout:10000});
    check(await p.evaluate(()=>window.chatTest.streams===3),'409 仅查询运行记录，不自动重新发送');
    check(await p.getByRole('button',{name:'新建会话',exact:true}).isEnabled(),'409 完成后不永久加载');
    await p.getByRole('link',{name:'完整助手',exact:true}).click();
    await p.setViewportSize({width:1440,height:900});
    await p.getByRole('button',{name:'最近会话',exact:true}).click();
    await p.getByRole('button',{name:'查看更多会话',exact:true}).click();
    await p.getByRole('button',{name:'删除会话：测试问题',exact:true}).click();
    await p.getByRole('status').filter({hasText:'会话已删除'}).waitFor();
    check(removed===1,'删除当前会话并显示通用提示');
    await p.getByRole('button',{name:'关闭会话列表'}).click();
    check(await p.getByRole('heading',{name:'你好，我是禾序 AI 助手'}).isVisible(),'删除后进入本地空白状态');
    return {passed:checks.length,checks};
  } finally { await context.close(); }
}
